        function math.clampAngle(angle,from,to)
            angle[1] = math.clamp(angle[1],from[1],to[1])
            angle[2] = math.clamp(angle[2],from[2],to[2])
            angle[3] = math.clamp(angle[3],from[3],to[3])
            
            return angle
        end
        
        
        function drawCrosshair(x,y,size)
            local matrix = Matrix()
            matrix:translate(Vector(x,y))
            matrix:setScale(Vector(size,size))
            
            render.pushMatrix(matrix)
            
            render.setColor(Color(200,200,200,80))
            render.drawCircle(0, 0, 12)
            
            render.setColor(Color(255,255,255,255))
            render.drawFilledCircle(0, 0, 2)
            
            render.popMatrix()
        end
        
        local w, h = render.getGameResolution()
        local lastScreenPos = Vector(w/2,h/2)
        local zoomSize = 1
        hook.add("DrawHUD", "", function()
            local tr = trace.line(left_arm:getPos(), left_arm:localToWorld(Vector(10000,0,0)), {base, seat, player()})
            
            local screenpos = tr.HitPos:toScreen()
            drawCrosshair(screenpos["x"], screenpos["y"], 1)
            
            tr = trace.line(right_arm:getPos(), right_arm:localToWorld(Vector(10000,0,0)), {base, seat, player()})
            
            screenpos = tr.HitPos:toScreen()
            drawCrosshair(screenpos["x"], screenpos["y"], 1)
            
            render.setColor(Color(200,200,200,80))
            tr = trace.line(seat:getPos(), seat:getPos() + player():getEyeAngles():getForward() * 10000, {base, seat, player()})

            render.drawRect(screenpos["x"]-15, screenpos["y"]-15, 30, 30)

        local zoom = 1
        
        hook.add( "CalcView", "DriverCamera", function()
            if not (isValid(base) and isValid(seat)) then return end
            if player() == seat:getDriver() then
                
                zoom = math.approach(zoom, player():keyDown(IN_KEY.ATTACK2) and 0 or 1, 0.4 * 10 * timer.frametime())
                
                local offAngle = Angle(math.clamp(15+eyeAngles()[1]/35,0,15),0,0)*math.easeInOutSine(zoom)
                local eyeAng = eyeAngles() + (thirdperson and offAngle or Angle())
                local camang = base:worldToLocalAngles(eyeAng)
                
                local thirdPersonTarget = (base:localToWorld(camera.pos + camera.pos*Vector(0,0,1)*0.2*(1-zoom) + Vector(-camera.distance,0,0):getRotated(camang)))
                
                if (not player():isTyping() or input.isGameUIVisible()) and input.isKeyDown(KEY.V) then
                    if down["v"] ~= true then
                        down["v"] = true
                        thirdperson = not thirdperson
                        lerpedCamPos = thirdPersonTarget
                    end
                else
                    down["v"] = false
                end
                
                if thirdperson then
                    lerpedCamPos = math.lerpVector((1 - (0.35 * zoom)) * 10 * timer.frametime(), lerpedCamPos, thirdPersonTarget)
                end
                
                return {
                    origin = thirdperson and lerpedCamPos or player():getEyePos()-seat:getRight()*12,
                    angles = math.clampAngle(eyeAng,-Angle(90,360,360),Angle(90,360,360)),
                    drawviewer = true,
                    fov = 100 - 35 * math.easeInOutSine(1-zoom)
                }
            end
        end)
    end)
