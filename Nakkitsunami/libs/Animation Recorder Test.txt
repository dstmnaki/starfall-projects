local write = true
local targetFPS = 45

local animationList = {
    --[[{
        name = "yamato_idle",
        realName = "wos_yamato_idle",
        fps = 60,
        frames = 60
    },--]]{
        name = "vergil_upperslash_1",
        realName = "wos_ymt_upperslash_1",
        fps = 45,
        frames = 120
    },
}
local animCount = #animationList

local animationIndex = 1
function writeAnimation()
    if animationIndex > #animationList then
        print("All done!")
        return
    end
    
    local animData = animationList[animationIndex]
    
    animationRecorder.recordAnimation(animData.name,targetFPS,write)
    animationIndex = animationIndex + 1
end
finishedFunction = function()
    timer.simple(0.1,function()
        writeAnimation()
    end)
end

function registerAnimations()
    timer.create("addAnim", 1/20, animCount, function()
        local reps = timer.repsleft("addAnim")
        local index = animCount-reps
        local animData = animationList[index]
        
        animationRecorder.addAnimation(animData.name,animData.realName,animData.frames,animData.fps,finishedFunction)
        if reps == 0 then
            print("Animations registered")
            timer.simple(0.1,function()
                print("Writing animations...")
                writeAnimation()
            end)
        end
    end)
end

registerAnimations()
