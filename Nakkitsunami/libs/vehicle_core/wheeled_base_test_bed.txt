local wheeled_base = require("Nakkitsunami/libs/vehicle_core/wheeled_base.txt")

local swayBarStrength = 0.65

local const = 50000
local damp = 2000

local suspension_height_front = 10
local suspension_height_back = 10

local suspension_length_front = 15
local suspension_length_back = 15

local suspension_width_front = 25
local suspension_width_back = 25

local suspension_forward_forward = 45
local suspension_forward_back = 45

local front_wheel_radius = 24
local rear_wheel_radius = 24

local data = {
    wheels = {
        ["left_front"] = {
            start_pos = function(base)
                if not isValid(base) then return Vector() end
                return base:localToWorld(Vector(suspension_forward_forward,suspension_width_front,suspension_height_front))
            end,
            end_pos = function(base)
                if not isValid(base) then return Vector() end
                return base:localToWorld(Vector(suspension_forward_forward,suspension_width_front,suspension_height_front-suspension_length_front-front_wheel_radius/2))
            end,
            constant = const,
            damping = damp,
            wheel_ent_data = function(ent_data)
                local sus = ent_data.suspensions["left_front"]
                local base_ent = ent_data.base_ent
                return {
                    sus.end_pos(sus.base),
                    front_wheel_radius,
                    {base = base_ent, suspension = sus}
                }
            end,
            steering_angle = 30,
            swayBarStrength = swayBarStrength
        },
        ["right_front"] = {
            start_pos = function(base)
                if not isValid(base) then return Vector() end
                return base:localToWorld(Vector(suspension_forward_forward,-suspension_width_front,suspension_height_front))
            end,
            end_pos = function(base)
                if not isValid(base) then return Vector() end
                return base:localToWorld(Vector(suspension_forward_forward,-suspension_width_front,suspension_height_front-suspension_length_front-front_wheel_radius/2))
            end,
            constant = const,
            damping = damp,
            wheel_ent_data = function(ent_data)
                local sus = ent_data.suspensions["right_front"]
                local base_ent = ent_data.base_ent
                return {
                    sus.end_pos(sus.base),
                    front_wheel_radius,
                    {base = base_ent, suspension = sus}
                }
            end,
            steering_angle = 30,
            swayBarStrength = swayBarStrength
        },
        ["left_back"] = {
            start_pos = function(base)
                if not isValid(base) then return Vector() end
                return base:localToWorld(Vector(-suspension_forward_back,suspension_width_back,suspension_height_back))
            end,
            end_pos = function(base)
                if not isValid(base) then return Vector() end
                return base:localToWorld(Vector(-suspension_forward_back,suspension_width_back,suspension_height_back-suspension_length_back-rear_wheel_radius/2))
            end,
            constant = const,
            damping = damp,
            wheel_ent_data = function(ent_data)
                local sus = ent_data.suspensions["left_back"]
                local base_ent = ent_data.base_ent
                return {
                    sus.end_pos(sus.base),
                    rear_wheel_radius,
                    {base = base_ent, suspension = sus}
                }
            end,
            handbrake_force_mult = 0.9,
            traction_mult_slow_speed_acceleration = 0.4,
            driveForceMult = 1,
            swayBarStrength = swayBarStrength
        },
        ["right_back"] = {
            start_pos = function(base)
                if not isValid(base) then return Vector() end
                return base:localToWorld(Vector(-suspension_forward_back,-suspension_width_back,suspension_height_back))
            end,
            end_pos = function(base)
                if not isValid(base) then return Vector() end
                return base:localToWorld(Vector(-suspension_forward_back,-suspension_width_back,suspension_height_back-suspension_length_back-rear_wheel_radius/2))
            end,
            constant = const,
            damping = damp,
            wheel_ent_data = function(ent_data)
                local sus = ent_data.suspensions["right_back"]
                local base_ent = ent_data.base_ent
                return {
                    sus.end_pos(sus.base),
                    rear_wheel_radius,
                    {base = base_ent, suspension = sus}
                }
            end,
            handbrake_force_mult = 0.9,
            traction_mult_slow_speed_acceleration = 0.4,
            driveForceMult = 1,
            swayBarStrength = swayBarStrength
        }
    }
}
wheeled_base(nil, nil, false, data, function(base_class, dt, ct)
    local base = base_class.base_ent
    
    local driver = seat:getDriver()
    local w, s, a, d = driver:getControls()
    
    local vel_len = base:getVelocity():getLength()
    local vel_local = base:getLocalVelocity()
    local vel_forward = vel_local.x
    local side_vel = vel_local.y/300
    
    local steer = math.approach(base_class.steer, (a and -1 or d and 1 or 0)/(1+math.easeInSine(math.clamp(vel_len/(150*5),0,1))*(4+math.abs(side_vel)))-math.clamp(side_vel,-1,1)*0.2, 0.3 * 10 * dt)
    base_class.steer = steer
    
    local handbrake = isValid(driver) and driver:keyDown(IN_KEY.JUMP)
    
    local isBraking = ((w and vel_forward < 0) or (s and vel_forward > 0))
    
    local brake_force = handbrake and 1 or isBraking and 0.05 or 0.015*(vel_len/300) 
    
    local throttle = math.approach(base_class.throttle, w and 1 or s and -0.5 or 0, 0.3 * 10 * dt)
    base_class.throttle = throttle
    
    base_class.wheels["left_front"].steer = -steer
    base_class.wheels["right_front"].steer = -steer
    
    local center_traction = Vector()
    local total_traction = 0
    local count = 0
    
    local traction_mult = (1-math.easeInExpo(math.max(0,math.abs(throttle)-math.abs(vel_forward)/200)))/(1+math.abs(side_vel)*math.abs(throttle))
    for k, wheel in pairs(base_class.wheels) do
        wheel.brake_force = brake_force
        wheel.handbrake_force = handbrake and wheel.handbrake_force_mult*(0.3+math.clamp(vel_len/100,0,1)*0.8) or 0
        wheel.traction = wheel.traction_mult_slow_speed_acceleration*(1-traction_mult)+traction_mult
        wheel.driveForce = 1000*wheel.driveForceMult*throttle
        
        if not wheel.isOnGround then continue end
        
        total_traction = total_traction + wheel.traction
        center_traction = center_traction + wheel.ground_pos
        count = count + 1
    end
    if count > 0 and not handbrake then
        center_traction = base:worldToLocal(center_traction / count)
        local center_target = base:localToWorld(center_traction*Vector(1,1,0.5))
        total_traction = total_traction / count
        base:applyForceOffsetFixed(base:getForward() * throttle * (500 / (1+math.easeInSine(math.clamp(vel_len/1000,0,1)))) * 1.25 * total_traction * base:getMass() * dt, center_target)
    end
end)
