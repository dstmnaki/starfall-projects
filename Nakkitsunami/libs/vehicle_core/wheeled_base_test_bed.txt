--@include Nakkitsunami/libs/vehicle_core/wheeled_base.txt
local wheeled_base = require("Nakkitsunami/libs/vehicle_core/wheeled_base.txt")


local suspension_height_front = 10
local suspension_height_back = 10

local suspension_length_front = 35
local suspension_length_back = 35

local suspension_width_front = 25
local suspension_width_back = 25

local suspension_forward = 45
local suspension_back = 45

local front_wheel_radius = 24
local rear_wheel_radius = 24


local driveForce = 1
local driveBiasFront = 0

local maxVel = 800


local allWheelSteering = false
local steeringLock = 30
local counterSteerMult = 0.5
local steeringRate = 0.3

local steeringReduceVel = 600
local steeringReduceAmount = 5

local suspensionBiasFront = 0.8
local suspensionStrength = 50000
local suspensionDamping = 3000

local antiRollBarBiasFront = 0.5
local antiRollMult = 0.65


local handbrakeBiasFront = 0.1
local handbrakeStrength = 0.9

local lowSpeedTractionLossMult = 0.4


local data = {
    parameterCallback = function()
        return {
            driveForce = driveForce,
            maxVel = maxVel,
            counterSteerMult = counterSteerMult,
            steeringRate = steeringRate,
            steeringReduceVel = steeringReduceVel,
            steeringReduceAmount = steeringReduceAmount,
        }
    end,
    wheels = {
        ["left_front"] = {
            start_pos = function(base)
                if not isValid(base) then return NO_POS end
                return base:localToWorld(Vector(suspension_forward,suspension_width_front,suspension_height_front))
            end,
            end_pos = function(base)
                if not isValid(base) then return NO_POS end
                return base:localToWorld(Vector(suspension_forward,suspension_width_front,suspension_height_front-suspension_length_front))
            end,
            constant = math.bias(suspensionBiasFront,suspensionStrength/2,suspensionStrength),
            damping = math.bias(suspensionBiasFront,suspensionDamping/2,suspensionDamping),
            wheel_ent_data = function(ent_data)
                local sus = ent_data.suspensions["left_front"]
                local base_ent = ent_data.base_ent
                return {
                    sus.end_pos(sus.base),
                    front_wheel_radius,
                    {base = base_ent, suspension = sus}
                }
            end,
            steering_angle = steeringLock,
            handbrake_force_mult = math.bias(handbrakeBiasFront,0,handbrakeStrength),
            driveForceMult = math.easeOutSine(driveBiasFront),
            swayBarStrength = math.bias(antiRollBarBiasFront,0,antiRollMult)
        },
        ["right_front"] = {
            start_pos = function(base)
                if not isValid(base) then return NO_POS end
                return base:localToWorld(Vector(suspension_forward,-suspension_width_front,suspension_height_front))
            end,
            end_pos = function(base)
                if not isValid(base) then return NO_POS end
                return base:localToWorld(Vector(suspension_forward,-suspension_width_front,suspension_height_front-suspension_length_front))
            end,
            constant = math.bias(suspensionBiasFront,suspensionStrength/2,suspensionStrength),
            damping = math.bias(suspensionBiasFront,suspensionDamping/2,suspensionDamping),
            wheel_ent_data = function(ent_data)
                local sus = ent_data.suspensions["right_front"]
                local base_ent = ent_data.base_ent
                return {
                    sus.end_pos(sus.base),
                    front_wheel_radius,
                    {base = base_ent, suspension = sus}
                }
            end,
            steering_angle = steeringLock,
            handbrake_force_mult = math.bias(handbrakeBiasFront,0,handbrakeStrength),
            driveForceMult = math.easeOutSine(driveBiasFront),
            swayBarStrength = math.bias(antiRollBarBiasFront,0,antiRollMult)
        },
        ["left_back"] = {
            start_pos = function(base)
                if not isValid(base) then return NO_POS end
                return base:localToWorld(Vector(-suspension_back,suspension_width_back,suspension_height_back))
            end,
            end_pos = function(base)
                if not isValid(base) then return NO_POS end
                return base:localToWorld(Vector(-suspension_back,suspension_width_back,suspension_height_back-suspension_length_back))
            end,
            constant = math.bias(1-suspensionBiasFront,suspensionStrength/2,suspensionStrength),
            damping = math.bias(1-suspensionBiasFront,suspensionDamping/2,suspensionDamping),
            wheel_ent_data = function(ent_data)
                local sus = ent_data.suspensions["left_back"]
                local base_ent = ent_data.base_ent
                return {
                    sus.end_pos(sus.base),
                    rear_wheel_radius,
                    {base = base_ent, suspension = sus}
                }
            end,
            steering_angle = allWheelSteering and -steeringLock or 0,
            handbrake_force_mult = math.bias(1-handbrakeBiasFront,0,handbrakeStrength),
            traction_mult_slow_speed_acceleration = lowSpeedTractionLossMult,
            driveForceMult = math.easeOutSine(1-driveBiasFront),
            swayBarStrength = math.bias(1-antiRollBarBiasFront,0,antiRollMult)
        },
        ["right_back"] = {
            start_pos = function(base)
                if not isValid(base) then return NO_POS end
                return base:localToWorld(Vector(-suspension_back,-suspension_width_back,suspension_height_back))
            end,
            end_pos = function(base)
                if not isValid(base) then return NO_POS end
                return base:localToWorld(Vector(-suspension_back,-suspension_width_back,suspension_height_back-suspension_length_back))
            end,
            constant = math.bias(1-suspensionBiasFront,suspensionStrength/2,suspensionStrength),
            damping = math.bias(1-suspensionBiasFront,suspensionDamping/2,suspensionDamping),
            wheel_ent_data = function(ent_data)
                local sus = ent_data.suspensions["right_back"]
                local base_ent = ent_data.base_ent
                return {
                    sus.end_pos(sus.base),
                    rear_wheel_radius,
                    {base = base_ent, suspension = sus}
                }
            end,
            steering_angle = allWheelSteering and -steeringLock or 0,
            handbrake_force_mult = math.bias(1-handbrakeBiasFront,0,handbrakeStrength),
            traction_mult_slow_speed_acceleration = lowSpeedTractionLossMult,
            driveForceMult = math.easeOutSine(1-driveBiasFront),
            swayBarStrength = math.bias(1-antiRollBarBiasFront,0,antiRollMult)
        }
    }
}

local base_ent = wheeled_base(pos, ang, frozen, data, function(base_class, dt, ct)
    local base = base_class.base_ent
    local driveForce = base_class.driveForce
    local steeringRate = base_class.steeringRate
    local steeringReduceVel = base_class.steeringReduceVel
    local steeringReduceAmount = base_class.steeringReduceAmount
    
    local driver = seat:getDriver()
    local w, s, a, d, space = driver:getControls()
    
    local vel_len = base:getVelocity():getLength()
    local vel_local = base:getLocalVelocity()
    local vel_forward = vel_local.x
    local side_vel = vel_local.y/300
    
    local steer = math.approach(base_class.steer, (a and -1 or d and 1 or 0)/(1+math.easeInSine(math.clamp(vel_len/(steeringReduceVel*2),0,1))*(steeringReduceAmount+math.abs(side_vel)))-math.clamp(side_vel,-1,1)*0.5*counterSteerMult, steeringRate * 10 * dt)
    base_class.steer = steer
    
    local handbrake = space
    
    local isBraking = ((w and vel_forward < 0) or (s and vel_forward > 0))
    
    local brake_force = isValid(driver) and (handbrake and 1 or (isBraking and 0.05 or 0.015*(vel_len/300))) or 1
    
    local throttle = math.approach(base_class.throttle, w and 1 or s and -0.5 or 0, 0.3 * 10 * dt)
    base_class.throttle = throttle
    
    base_class.wheels["left_front"].steer = -steer
    base_class.wheels["right_front"].steer = -steer
    
    base_class.wheels["left_back"].steer = -steer
    base_class.wheels["right_back"].steer = -steer
    
    local center_traction = Vector()
    local total_traction = 0
    local count = 0
    
    local traction_mult = (1-math.easeInExpo(math.max(0,math.abs(throttle)-math.abs(vel_forward)/200)))/(1+math.abs(side_vel)*math.abs(throttle))
    for k, wheel in pairs(base_class.wheels) do
        wheel.brake_force = brake_force
        wheel.handbrake_force = handbrake and wheel.handbrake_force_mult*(0.3+math.clamp(vel_len/100,0,1)*0.8) or 0
        wheel.traction = ((wheel.traction_mult_slow_speed_acceleration*(1-traction_mult)+traction_mult))*(1-0.4*math.clamp(math.abs(vel_forward/50),0,1))
        wheel.driveForce = 1000*wheel.driveForceMult*throttle
        
        if not wheel.isOnGround then continue end
        
        total_traction = total_traction + wheel.traction/0.6
        center_traction = center_traction + wheel.ground_pos
        count = count + 1
    end
    if count > 0 and not handbrake then
        center_traction = base:worldToLocal(center_traction / count)
        local center_target = base:localToWorld(center_traction*Vector(0.25,0.25,1))
        total_traction = total_traction / count
        base:applyForceOffsetFixed(base:getForward() * throttle * driveForce * (500 / (1+math.easeInSine(math.clamp(vel_len/(maxVel/800*1000),0,1)))) * 1.25 * total_traction * base:getMass() * dt, center_target)
    end
end)
