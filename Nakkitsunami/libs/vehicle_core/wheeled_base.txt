require("Nakkitsunami/libs/vehicle_core/_shared.txt")
local suspension = require("Nakkitsunami/libs/vehicle_core/suspension.txt")
local wheel = require("Nakkitsunami/libs/vehicle_core/wheel.txt")

local wheeled_base = class("C_WHEELED_BASE")
local wheeled_bases = {}
local wheeled_basesID = 1

function wheeled_base:initialize(pos, ang, frozen, data, think)
    pos, ang, frozen = pos or chip():getPos()+Vector(0,0,45), ang or chip():getAngles()*Angle(0,1,0), frozen or false
    data = data or {}
    
    local base_model = data.base_model or "models/sprops/cuboids/height12/size_4/cube_48x108x12.mdl"
    
    local base_ent = prop.create(pos, ang, base_model, frozen)
    base_ent:setMass(data.mass or 1000)
    base_ent:setInertia(data.inertia or Vector(500))
    
    base_ent:setMaterial("models/wireframe")
    
    local seat_pos = data.seat_pos or base_ent:localToWorld(Vector(0,0,5))
    local seat_ang = data.seat_ang or base_ent:localToWorldAngles(Angle(0,-90,0))
    local seat_parent = data.seat_parent or base_ent
    
    seat = prop.createSeat(seat_pos, seat_ang, "models/nova/airboat_seat.mdl", true)
    seat:setParent(isfunction(seat_parent) and seat_parent(base_ent) or seat_parent)
    seat:doNotDuplicate(false)
    
    self.seat = seat
    self.base_ent = base_ent
    
    local wheels = data.wheels
    
    self.wheels = {}
    self.suspensions = {}
    
    for wheel_index, wheel_data in pairs(wheels) do
        local start_pos = wheel_data.start_pos
        local end_pos = wheel_data.end_pos
        local constant = wheel_data.constant
        local damping = wheel_data.damping
        
        local sus = suspension.create(base_ent, start_pos, end_pos, constant, damping)
        
        self.suspensions[wheel_index] = sus
        
        local wheel_ent = wheel.create(
            unpack(wheel_data.wheel_ent_data(self))
        )
        
        wheel_ent.handbrake_force_mult = wheel_data.handbrake_force_mult or 0.15
        wheel_ent.traction_mult_slow_speed_acceleration = wheel_data.traction_mult_slow_speed_acceleration or 1
        
        wheel_ent.driveForceMult = wheel_data.driveForceMult or 0
        
        wheel_ent.steering_angle = wheel_data.steering_angle or 0
        
        wheel_ent.swayBarStrength = wheel_data.swayBarStrength or 0
        
        sus.swayBarStrength = wheel_data.swayBarStrength or 0
        
        wheel_ent.wheelwell_pos = start_pos
        
        sus.wheel = wheel_ent
        
        self.wheels[wheel_index] = wheel_ent
    end
    
    self.think = think or function() end
    
    self.steer = 0
    self.throttle = 0
    
    self.id = wheeled_basesID
    wheeled_bases[wheeled_basesID] = self
    wheeled_basesID = wheeled_basesID + 1
    
    return self
end

local lastThink = timer.curtime()
local function delta()
    local ct = timer.curtime()
    local delta = ct - lastThink
    lastThink = ct
    return delta, ct
end
timer.create("wheeled_base_main",1/20,0,function()
    local dt, ct = delta()
    for _, wheeled_base in pairs(wheeled_bases) do
        wheeled_base.think(wheeled_base, dt, ct)
    end
end)

wheeled_base.create = wheeled_base

return wheeled_base
