local ENTITY = getMethods("Entity")
local m_in_sq = 1 / 39.37 ^ 2 -- in^2 to m^2
local const = m_in_sq * 360 / (2 * 3.1416)
function ENTITY:applyForceOffsetFixed(force, pos)
    if CLIENT then return end
    self:applyForceCenter(force)
    
    local off = pos - self:localToWorld(self:getMassCenter())
    local angf = off:cross(force) * const
    
    self:applyTorque(angf)
end

function ENTITY:getControls()
    if not isValid(self) then return false, false, false, false, false end
    local w = isValid(self) and self:keyDown(IN_KEY.FORWARD)
    local s = isValid(self) and self:keyDown(IN_KEY.BACK)
    local a = isValid(self) and self:keyDown(IN_KEY.MOVELEFT)
    local d = isValid(self) and self:keyDown(IN_KEY.MOVERIGHT)
    
    local space = isValid(self) and self:keyDown(IN_KEY.JUMP)
    
    return w, s, a, d, space
end

POS_FORWARD = Vector(1,0,0)
POS_RIGHT = Vector(0,1,0)
POS_UP = Vector(0,0,1)

POS_NO_FORWARD = POS_RIGHT+POS_UP
POS_NO_RIGHT = POS_FORWARD+POS_UP
POS_NO_UP = POS_FORWARD+POS_RIGHT

XYZ = POS_NO_FORWARD+POS_FORWARD
NO_POS = POS_FORWARD*0

ANG_PITCH = Angle(1,0,0)
ANG_YAW = Angle(0,1,0)
ANG_ROLL = Angle(0,0,1)

ANG_NO_PITCH = ANG_YAW+ANG_ROLL
ANG_NO_YAW = ANG_PITCH+ANG_ROLL
ANG_NO_ROLL = ANG_PITCH+ANG_YAW

PYR = ANG_NO_PITCH+ANG_PITCH
NO_ANG = ANG_PITCH*0
